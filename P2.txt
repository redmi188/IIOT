import RPi.GPIO as GPIO
import time

# Use BCM numbering (GPIO numbers, NOT physical pin numbers)
GPIO.setmode(GPIO.BCM)

# Define GPIO pins
GPIO_TRIGGER = 23   # GPIO23 (Pin 16 on board)
GPIO_ECHO = 24      # GPIO24 (Pin 18 on board)

# Set pin directions
GPIO.setup(GPIO_TRIGGER, GPIO.OUT)  # Trigger pin as output
GPIO.setup(GPIO_ECHO, GPIO.IN)      # Echo pin as input

def distance():
    # Send a short pulse to trigger ultrasonic burst
    GPIO.output(GPIO_TRIGGER, True)
    time.sleep(0.00001)  # 10 microseconds pulse
    GPIO.output(GPIO_TRIGGER, False)

    # Initialize start and stop times
    start_time = time.time()
    stop_time = time.time()

    # Wait for echo to go HIGH (sound wave leaves sensor)
    while GPIO.input(GPIO_ECHO) == 0:
        start_time = time.time()

    # Wait for echo to go LOW (sound wave returns)
    while GPIO.input(GPIO_ECHO) == 1:
        stop_time = time.time()

    # Calculate time difference
    time_elapsed = stop_time - start_time

    # Calculate distance:
    # Speed of sound = 34300 cm/s
    # Divide by 2 because signal travels forth and back
    distance = (time_elapsed * 34300) / 2

    return distance

try:
    while True:
        dist = distance()
        print(f"Measured Distance = {dist:.1f} cm")
        time.sleep(1)

# Clean up GPIO when program is stopped
except KeyboardInterrupt:
    GPIO.cleanup()
