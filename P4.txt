# Import time module for delay
import time

# board library helps map Raspberry Pi pins (like D4 → GPIO4)
import board

# Library to communicate with DHT11 sensor
import adafruit_dht


# Create DHT11 sensor object
# board.D4 means GPIO4 (Physical Pin 7)
# use_pulseio=False is used on Raspberry Pi to avoid timing issues
dht = adafruit_dht.DHT11(board.D4, use_pulseio=False)


# Infinite loop to continuously read sensor data
while True:
    try:
        # Read temperature from sensor (in Celsius)
        temperature = dht.temperature

        # Read humidity from sensor (in %)
        humidity = dht.humidity

        # Print temperature with 1 decimal point
        print(f"Temperature: {temperature:.1f} °C")

        # Print humidity with 1 decimal point
        print(f"Humidity: {humidity:.1f} %")

        # Separator for better readability
        print("----------------------")

    except RuntimeError as e:
        # DHT11 sometimes gives errors due to timing issues
        # This is normal → we just ignore and retry
        print("Retrying:", e)

    # Wait 2 seconds before next reading
    # (DHT11 is slow sensor, needs delay)
    time.sleep(2)
